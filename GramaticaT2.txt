INICIO -> GENERARREPORTE

GENERARREPORTE ->"GenerarReporteEstadistico" "{" INSTRUCCIONES "}"

INSTRUCCIONES -> INSTRUCCIONES INSTRUCCION
                | INSTRUCCION

INSTRUCCION -> COMPARE
                |DEFINIRGLOBALES
                |GRAFICABARRAS
                |GRAFICAPIE
                |GRAFICALINEAL

COMPARE -> "(" "'" cadena "'" "," "'" cadena "'" ")" ";"

DEFINIRGLOBALES -> "definirglobales" "{" DEFINICIONVARIABLESG "}"
                
DEFINICIONVARIABLESG -> DEFINICIONVARIABLESG DEFINICIONVARIABLEG
                |DEFINICIONVARIABLEG

DEFINICIONVARIABLEG -> TIPO id "=" VALORTIPO ";"
        |TIPO

TIPO -> string
        |double

VALORTIPO -> cadena
        |digito
        |"$" "{" id "," cadena "," cadena "," "cadena" "}"

GRAFICABARRAS -> "graficabarras" "{" INFOSBARRAS "}"

GRAFICASLINEAS -> "graficaLineas" "{" INFOSLINEAS "}"

GRAFICAPIE -> "graficapie" "{" INFOS "}" 

INFOSBARRAS ->INFOSBARRAS "titulo" ":" id ";"
        |INFOSBARRAS "ejex" ":" "[" DATOS "]"
        |INFOSBARRAS VALORES
        |INFOSBARRAS "tituloX" ":" cadena ";"
        |INFOSBARRAS "tituloY" ":" cadena ";"
        |"titulo" ":" id ";"
        |"ejex" ":" "[" DATOS "]"
        |"tituloX" ":" cadena ";"
        |"tituloY" ":" cadena ";"
        |VALORES

INFOSLINEAS -> INFOSLINEAS "titulo" ":" id ";"
        | "Archivo" ":" cadena ";"

INFOS-> INFOS "titulo" ":" id ";"
        |INFOS "ejex" ":" "[" DATOS "]"
        |INFOS VALORES
        |"titulo" ":" id ";"
        |"ejex" ":" "[" DATOS "]"
        |VALORES

VALORES -> "valores" ":" "[" DATOS "]"

DATOS -> DATOS "," DATO
        |DATO

DATO -> decimal
        |digito
        |cadena
        |id
        |"$" "{" id "," cadena "," cadena "," "cadena""}"
