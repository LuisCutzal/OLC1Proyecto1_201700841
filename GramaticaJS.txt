INICIO::= CLASES
    ;
CLASES::= CLASES CLASE
    |CLASE
    ;
CLASES::= class_js id parentesisA parentesisC llaveabre INSTRUCCIONES llavecierra
    | class_js id llaveabre INSTRUCCIONES llavecierra
    ;


INSTRUCCIONES::=INSTRUCCIONES INSTRUCCION
    |INSTRUCCION
    ;


INSTRUCCION::=IMPORT
    |METODO
    |DECLARACIONVARIABLES
    |ASIGNACIONVARIABLES
    |SENTENCIAIF
    |SENTENCIAFOR   
    |SENTENCIAWHILE
    |SENTENCIADOWHILE
    |SENTENCIASWITCH
    |CONSOLA
    |BREAK
    |LLAMADA
    ;

IMPORT::= TIPO id igual require parentesisA EXPRESION parentesisC puntoycoma
    | TIPO id igual require parentesisA EXPRESION parentesisC
    ;

METODO::= id parentesisA PARAMETROS parentesisC llaveabre INSTRUCCIONESMETODO llavecierra
    | id parentesisA  parentesisC llaveabre INSTRUCCIONESMETODO llavecierra
    ;

PARAMETROS::= PARAMETROS coma id
    |id
    ;
INSTRUCCIONESMETODO::= INSTRUCCIONESMETODO INSTRUCCIONMETODO
    |INSTRUCCIONMETODO
    ;

INSTRUCCIONMETODO::= DECLARACIONVARIABLES
    |ASIGNACIONVARIABLES
    |SENTENCIAIF
    |SENTENCIAFOR   
    |SENTENCIAWHILE
    |SENTENCIADOWHILE
    |SENTENCIASWITCH
    |CONSOLA
    |BREAK
    |LLAMADA
    ;

DECLARACIONVARIABLES::= TIPO id igual EXPRESION puntoycoma
    |TIPO id igual EXPRESION
    |TIPO id puntoycoma
    |TIPO id igual id puntoycoma
    |TIPO id igual id
    |TIPO id 
    ;

EXPRESION::= EXPRESION igualigual EXPRESION 
    |EXPRESION diferente EXPRESION
    |EXPRESION menorque EXPRESION
    |EXPRESION mayorque EXPRESION
    |EXPRESION menorIgual EXPRESION
    |EXPRESION mayorIgual EXPRESION
    |EXPRESION AND EXPRESION
    |EXPRESION OR EXPRESION
    |NOT EXPRESION
    |EXPRESION mas EXPRESION
    |EXPRESION menos EXPRESION
    |EXPRESION multiplicacion EXPRESION
    |EXPRESION division EXPRESION
    |EXPRESION potencia EXPRESION
    |EXPRESION modulo EXPRESION
    |menos EXPRESION
    |true_js
    |false_js
    |cadena
    |numero
    |decimal
    |caracter
    ;

ASIGNACIONVARIABLES::= id igual EXPRESION puntoycoma
    |id igual EXPRESION
    ;

SENTENCIAIF::= if_js parentesisA EXPRESION parentesisC llaveabre INSTRUCCIONESMETODO llavecierra
    | if_js parentesisA EXPRESION parentesisC llaveabre INSTRUCCIONESMETODO llavecierra else_js SENTENCIAIF
    | if_js parentesisA EXPRESION parentesisC llaveabre INSTRUCCIONESMETODO llavecierra else_js llaveabre INSTRUCCIONESMETODO llavecierra
    ;

SENTENCIAFOR::= for_js parentesisA DECLARACIONFOR puntoycoma EXPRESION puntoycoma CONTADOR parentesisC llaveabre INSTRUCCIONESMETODO llavecierra
    ;

DECLARACIONFOR::= TIPO id igual EXPRESION
    | id igual EXPRESION
    ;

CONTADOR::= id INCREMENTO
    ;
INCREMENTO::= incrementoUno
    |decrementoUno
    |mas numero
    |menos numero
    ;

TIPO::= string_js
    |double_js
    |bool_js
    |int_js
    ;

SENTENCIAWHILE::= while_js parentesisA EXPRESION parentesisC llaveabre INSTRUCCIONESMETODO llavecierra
    ;

SENTENCIADOWHILE::= do_js llaveabre INSTRUCCIONESMETODO llavecierra while_js parentesisA EXPRESION parentesisC puntoycoma
    | do_js llaveabre INSTRUCCIONESMETODO llavecierra while_js parentesisA EXPRESION parentesisC 
    ;

SENTENCIASWITCH::= switch_js parentesisA EXPRESION parentesisC llaveabre CASOS llavecierra
    ;

CASOS::= case_js EXPRESION dospuntos INSTRUCCIONESMETODO
        |default dospuntos INSTRUCCIONESMETODO
    ;

CONSOLA::=console_js punto log_js parentesisA EXPRESION parentesisC puntoycoma
    ;

BREAK::= break_js puntoycoma
    | break_js
    ;

LLAMADA::= id parentesisA parentesisC puntoycoma
    |id parentesisA parentesisC
    |id parentesisA LISTAPARAMETROS parentesisC puntoycoma
    |id parentesisA LISTAPARAMETROS parentesisC 
    ;

LISTAPARAMETROS::= LISTAPARAMETROS coma EXPRESION
    |EXPRESION
    ;
