INICIO -> CLASES
//**********CLASES*********

CLASES -> CLASES CLASE
    | CLASE

CLASE -> "class" id "(" "") "{" INSTRUCCIONES "}"
        | "class" id "{" INSTRUCCIONES "}"

//**********INSTRUCCIONES*********

INSTRUCCIONES -> INSTRUCCIONES -> INSTRUCCION
    |INSTRUCCION

INSTRUCCION -> IMPORT
        |METODO
        |DECLARACIONVARIABLES
        |ASIGNACIONVARIABLES
        |SENTENCIAIF
        |SENTENCIAFOR
        |SENTENCIAWHILE
        |SENTENCIADOWHILE
        |SENTENCIASWITCH
        |CONSOLA
        |BREAK
        |LLAMADA

//**********IMPORT*********

IMPORT -> TIPO id igual require ( EXPRESION ) pyc
        | TIPO id igual require ( EXPRESION )

//**********METODO*********

METODO -> id "(" PARAMETROS ")" "{" INSTRUCCIONESMETODO "}"
        | id "(" ")" "{" INSTRUCCIONESMETODO "}"

PARAMETROS -> PARAMETROS coma id
        |id


INSTRUCCIONESMETODO -> INSTRUCCIONESMETODO INSTRUCCIONMETODO
        |INSTRUCCIONMETODO

INSTRUCCIONMETODO ->  DECLARACIONVARIABLES
        | ASIGNACIONVARIABLES
        | SENTENCIAIF
        | SENTENCIAFOR
        | SENTENCIAWHILE
        | SENTENCIADOWHILE
        | SENTENCIADOWHILE
        | SENTENCIASWITCH
        | CONSOLA
        | BREAK
        | LLAMADA

//**********DECLARACIONVARIABLES*********

DECLARACIONVARIABLES-> TIPO id "=" EXPRESION ";"
        | TIPO id "=" EXPRESION
        | TIPO id ";"
        | TIPO id

EXPRESION-> EXPRESION "==" EXPRESION
        |EXPRESION "!=" EXPRESION
        |EXPRESION "<" EXPRESION
        |EXPRESION ">" EXPRESION
        |EXPRESION "<=" EXPRESION
        |EXPRESION ">=" EXPRESION
        |EXPRESION "&&" EXPRESION
        |EXPRESION "||" EXPRESION
        | "!" EXPRESION
        |EXPRESION "+" EXPRESION
        |EXPRESION "-" EXPRESION
        |EXPRESION "*" EXPRESION
        |EXPRESION "/" EXPRESION
        |EXPRESION "**" EXPRESION
        |EXPRESION "%" EXPRESION
        |"-" EXPRESION
        | true | fasle | id | cadena| digito | decimal | caracter


//**********ASIGNACIONVARIABLES*********

ASIGNACIONVARIABLES -> id "=" EXPRESION ";"
                    | id "=" EXPRESION 

//**********SENTENCIAIF*********

SENTENCIAIF-> if "(" EXPRESION ")" "{" INSTRUCCIONESMETODO "}"
            | if "(" EXPRESION ")" "{" INSTRUCCIONESMETODO "}" else SENTENCIAIF
            | if "(" EXPRESION ")" "{" INSTRUCCIONESMETODO "}" else "{" INSTRUCCIONESMETODO "}"


//**********SENTENCIAFOR*********

SENTENCIAFOR -> for "(" DECLARACIONFOR ";" EXPRESION ";" CONTADOR ")" "{" INSTRUCCIONESMETODO "}"

DECLARACIONFOR -> TIPO id "=" EXPRESION
                    |id "=" EXPRESION

CONTADOR -> id INCREMENTO

INCREMENTO -> "++"
            |"--"
            |"+"digito
            |"-"digito


TIPO -> string
        |double
        |bool
        |int

//**********SENTENCIAWHILE*********

SENTENCIAWHILE -> while "(" EXPRESION ")" "{" INSTRUCCIONESMETODO "}"

//**********SENTENCIADOWHILE*********

SENTENCIADOWHILE -> do "{" INSTRUCCIONESMETODO "}" while "(" EXPRESION ")" ";"
                | do "{" INSTRUCCIONESMETODO "}" while "(" EXPRESION ")"

//**********SENTENCIASWITCH*********

SENTENCIASWITCH -> switch "(" EXPRESION ")" "{" CASOS "}"

CASOS -> case EXPRESION ":" INSTRUCCIONESMETODO

//**********CONSOLA*********

CONSOLA -> console "." log "(" EXPRESION ")"

//**********BREAK*********

BREAK -> break ";"
        |break

//**********LLAMADA*********

LLAMADA -> id "(" ")" ";"
        | id "(" ")"
        | id "(" LISTAPARAMETROS ")" ";"
        | id "(" LISTAPARAMETROS ")"

LISTAPARAMETROS -> LISTAPARAMETROS "," EXPRESION
        |EXPRESION
